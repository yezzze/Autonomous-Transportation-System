### 1️⃣ State 定义对比

#### 原始版本 (`src/graph/types.py`)

class State(MessagesState):
    """State for the agent system"""

#### 分布式版本 (`src/graph/distributed_types.py`)
```python
class DistributedState(MessagesState):
    """分布式 Agent 调度器的 State"""
    
    # ========== 新增：L3 Agent 注册表 ==========

    timeout_seconds: int                   # 🆕 超时时间

**关键变化**：
- ❌ 删除 `TEAM_MEMBERS`（静态配置）
- ✅ 新增 `agent_registry_cache`（动态查询）
- ✅ 新增 `execution_plan`（包含 IP 地址）

### 2️⃣ Planner 节点对比

#### 原始版本 (`src/graph/nodes.py`)
```python
def planner_node(state: State) -> Command[Literal["supervisor", "__end__"]]:
    """Planner node that generate the full plan."""
    
    # 1. 应用 Prompt 模板
    messages = apply_prompt_template("planner", state)
    
    # 2. 选择 LLM
    llm = get_llm_by_type("basic")
    if state.get("deep_thinking_mode"):
        llm = get_llm_by_type("reasoning")
    
    # 3. 可选：搜索前置知识
    if state.get("search_before_planning"):
        searched_content = tavily_tool.invoke({"query": state["messages"][-1].content})
        messages[-1].content += f"\n\n# Search Results\n\n{searched_content}"
    
    # 4. 调用 LLM
    stream = llm.stream(messages)
    full_response = ""
    for chunk in stream:
        full_response += chunk.content
    
    # 5. 清理 JSON
    if full_response.startswith("```json"):
        full_response = full_response.removeprefix("```json")
    if full_response.endswith("```"):
        full_response = full_response.removesuffix("```")
    
    # 6. 验证 JSON
    goto = "supervisor"
    try:
        json.loads(full_response)
    except json.JSONDecodeError:
        goto = "__end__"
    
    # 7. 返回
    return Command(
        update={
            "messages": [HumanMessage(content=full_response, name="planner")],
            "full_plan": full_response,
        },
        goto=goto,
    )
```

#### 分布式版本 (`src/graph/distributed_nodes.py`)
```python
def distributed_planner_node(state: DistributedState) -> Command[Literal["executor", "__end__"]]:
    """分布式规划器节点"""
    
    # 🆕 1. 查询 L3 Agent 注册表
    registry_client = get_registry_client()
    available_agents = registry_client.query_agents()
    
    if not available_agents:
        return Command(goto="__end__", update={"plan_generated": False})
    
    # 🆕 2. 构建 Agent 能力描述
    agent_capabilities_desc = "\n".join([
        f"- **{agent['id']}** (IP: {agent['ip']}:{agent['port']}): {agent['description']}"
        for agent in available_agents
    ])
    
    # 🆕 3. 构建新的 System Prompt（关键变化！）
    system_prompt = f"""
你是一个分布式任务调度规划器。

## 可用的 L3 Agent 列表
{agent_capabilities_desc}

## 输出格式
{{
  "thought": "你的推理",
  "tasks": [
    {{
      "task_id": "task_001",
      "task_title": "任务标题",
      "task_description": "详细描述",
      "assigned_agent_id": "search_agent_001",  // ← 必须匹配注册表中的 ID
      "capability_required": "search"
    }}
  ]
}}
"""
    
    messages = [
        SystemMessage(content=system_prompt),
        HumanMessage(content=f"用户请求：{state['messages'][-1].content}")
    ]
    
    # ✅ 4. 调用 LLM（复用）
    llm = get_llm_by_type("reasoning")
    response = llm.invoke(messages)
    plan_json_str = response.content
    
    # ✅ 5. 清理 JSON（复用）
    if plan_json_str.startswith("```json"):
        plan_json_str = plan_json_str.removeprefix("```json")
    # ...
    
    # ✅ 6. 解析 JSON（复用）
    plan_data = json.loads(plan_json_str)
    
    # 🆕 7. 将 Agent ID 映射到 IP 地址（核心新增！）
    execution_plan: list[TaskAssignment] = []
    for task in plan_data.get("tasks", []):
        agent_id = task.get("assigned_agent_id")
        agent_info = registry_client.get_agent_by_id(agent_id)  # 🆕 查询注册表
        
        if not agent_info:
            logger.warning(f"Agent {agent_id} 不存在")
            continue
        
        task_assignment: TaskAssignment = {
            "task_id": task.get("task_id"),
            "task_title": task.get("task_title"),
            "task_description": task.get("task_description"),
            "assigned_agent_id": agent_id,
            "target_ip": agent_info["ip"],        # 🆕 补充 IP
            "target_port": agent_info["port"],    # 🆕 补充端口
            "status": "pending",                   # 🆕 初始状态
            "result": "",                          # 🆕 结果字段
            "retry_count": 0                       # 🆕 重试计数
        }
        execution_plan.append(task_assignment)
    
    # 🆕 8. 返回执行计划
    return Command(
        update={
            "execution_plan": execution_plan,      # 🆕 核心输出
            "current_task_index": 0,               # 🆕 初始化索引
            "plan_generated": True,
            "agent_registry_cache": available_agents,
        },
        goto="executor"  # 🆕 跳转到执行器（不是 supervisor）
    )
```

**关键变化总结**：

| 方面 | 原始版本 | 分布式版本 |
|------|---------|-----------|
| 输入 | 用户请求 | 用户请求 + L3 注册表 |
| Prompt | 要求输出工具调用 | 要求输出 Agent ID |
| 输出 | JSON 计划（steps） | TaskAssignment 列表（含 IP） |
| 下一步 | supervisor | executor |

```python
def distributed_executor_node(state: DistributedState) -> Command[Literal["monitor", "__end__"]]:
    """分布式执行器节点 - 统一的远程调用"""
    import requests
    
    # 🆕 1. 获取当前任务
    current_index = state.get("current_task_index", 0)
    execution_plan = state["execution_plan"]
    
    if current_index >= len(execution_plan):
        return Command(goto="monitor", update={"all_tasks_completed": True})
    
    current_task = execution_plan[current_index]
    
    # 🆕 2. 构建远程调用请求（核心变化！）
    target_url = f"http://{current_task['target_ip']}:{current_task['target_port']}/execute"
    
    payload = {
        "task_id": current_task["task_id"],
        "task_description": current_task["task_description"],
        "context": {
            "user_request": state["messages"][0].content,
            "previous_results": [
                task["result"] for task in execution_plan[:current_index]
                if task["status"] == "completed"
            ]
        }
    }
    
    timeout = state.get("timeout_seconds", 30)
    
    # 🆕 3. 执行远程调用（完全新增！）
    try:
        logger.info(f"发送请求到 {target_url}")
        
        # ⚠️ Stub 实现 - 生产环境需启用真实调用
        # response = requests.post(target_url, json=payload, timeout=timeout)
        # response.raise_for_status()
        # result_data = response.json()
        
        # 🔧 Mock 响应
        result_data = {
            "status": "success",
            "task_id": current_task["task_id"],
            "result": f"[Mock] Agent {current_task['assigned_agent_id']} 完成任务"
        }
        
        task_status = "completed"
        result_message = result_data.get("result")
        
    except requests.Timeout:
        task_status = "failed"
        result_message = f"错误：超时（{timeout}秒）"
    except requests.ConnectionError:
        task_status = "failed"
        result_message = f"错误：无法连接到 {target_url}"
    except Exception as e:
        task_status = "failed"
        result_message = f"错误：{str(e)}"
    
    # 🆕 4. 更新任务状态
    updated_plan = state["execution_plan"].copy()
    updated_plan[current_index]["status"] = task_status
    updated_plan[current_index]["result"] = result_message
    
    # 🆕 5. 移动到下一个任务
    next_index = current_index + 1
    
    return Command(
        update={
            "messages": [HumanMessage(
                content=f"### 任务结果 ({current_index+1}/{len(execution_plan)})\n{result_message}",
                name="executor"
            )],
            "execution_plan": updated_plan,
            "current_task_index": next_index
        },
        goto="monitor"  # 🆕 跳转到监控节点
    )
```

**关键变化总结**：

| 方面 | 原始版本 | 分布式版本 |
|------|---------|-----------|
| 节点数量 | 3个专门节点 | 1个通用节点 |
| 执行方式 | `research_agent.invoke()` | `requests.post()` |
| 工具集 | tavily, python_repl, browser | 无（远程调用）|
| 错误处理 | ReAct Agent 内部处理 | 显式的 try-except + 重试 |
| 下一步 | 固定到 supervisor | 到 monitor（循环控制）|

### 4️⃣ Supervisor/Monitor 对比

#### 原始版本 (`src/graph/nodes.py`)
```python
def supervisor_node(state: State) -> Command[Literal[*TEAM_MEMBERS, "__end__"]]:
    """Supervisor node that decides which agent should act next."""
    
    # 1. 应用 Prompt
    messages = apply_prompt_template("supervisor", state)
    
    # 2. 使用 LLM + 结构化输出决策
    response = (
        get_llm_by_type(AGENT_LLM_MAP["supervisor"])
        .with_structured_output(Router)  # 🔧 LLM 决策
        .invoke(messages)
    )
    
    goto = response["next"]  # "researcher" | "coder" | "browser" | "reporter" | "FINISH"
    
    if goto == "FINISH":
        goto = "__end__"
    
    return Command(goto=goto)
```

#### 分布式版本 (`src/graph/distributed_nodes.py`)
```python
def distributed_monitor_node(state: DistributedState) -> Command[Literal["executor", "reporter", "__end__"]]:
    """监控节点 - 简化的路由逻辑"""
    
    # 🆕 简单的逻辑判断（不需要 LLM！）
    current_index = state.get("current_task_index", 0)
    total_tasks = len(state.get("execution_plan", []))
    
    if current_index < total_tasks:
        # 还有任务待执行
        return Command(goto="executor")
    else:
        # 所有任务完成
        return Command(
            update={"all_tasks_completed": True},
            goto="reporter"
        )
```

**关键变化总结**：

| 方面 | 原始版本（Supervisor） | 分布式版本（Monitor） |
|------|----------------------|---------------------|
| 决策方式 | LLM 推理 | 简单逻辑判断 |
| 路由选项 | 4-5个节点 | 2个节点（executor/reporter）|
| Token 消耗 | 每次决策消耗 Token | 零 Token |
| 灵活性 | 高（可动态调整）| 低（固定流程）|

## 📊 整体架构对比

### 原始 LangManus 架构
```
用户 → Coordinator → Planner → Supervisor ⇄ [Researcher, Coder, Browser] → Reporter
                                   ↓
                           [tavily, python_repl, browser_tool]
```

### 分布式架构
```
用户 → Planner → Executor → Monitor → Reporter
       ↓          ↓          ↑ ↓
    [L3 Registry] [HTTP POST] └─ 循环
                   ↓
              [远程 L3 Agent]
```

## 🎯 核心改变一览表

| 组件 | 删除 ❌ | 保留 ✅ | 新增 🆕 |
|------|--------|--------|--------|
| **State** | `TEAM_MEMBERS`, `deep_thinking_mode` | `MessagesState`, `next` | `agent_registry_cache`, `execution_plan` |
| **Planner** | 本地搜索（tavily） | LLM 调用结构 | L3 注册表查询、IP 映射 |
| **Executor** | ReAct Agent、本地工具 | Command 模式 | HTTP 客户端、重试机制 |
| **Supervisor** | LLM 决策 | 路由概念 | 简化为逻辑判断（Monitor）|
| **Tools** | tavily, python_repl, browser | - | agent_registry |
| **Graph** | coordinator 节点 | StateGraph API | 线性流程（planner→executor→monitor）|
